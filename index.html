In your Google Sheet → Extensions → Apps Script → replace all code with this:

const SHEET_ID = '1RCZTOvlqDL14eaZlfS75zmkDnHo56NEYe5q3fm_U_gg'; function doPost(e) { const d = JSON.parse(e.postData.contents); const ss = SpreadsheetApp.openById(SHEET_ID); const sh = ss.getSheetByName(d.sheet); if (!sh) return out({ok:false, err:'Sheet not found: '+d.sheet}); if (d.action === 'append') { sh.appendRow(d.row); return out({ok:true}); } if (d.action === 'delete') { sh.deleteRow(d.sheetRow); return out({ok:true}); } return out({ok:false, err:'Unknown action'}); } function out(obj) { return ContentService.createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); }
