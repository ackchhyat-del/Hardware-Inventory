Apps Script Setup

Copy this into your Google Sheet's Apps Script editor:

const SHEET_ID = '1RCZTOvlqDL14eaZlfS75zmkDnHo56NEYe5q3fm_U_gg'; function doPost(e) { const d = JSON.parse(e.postData.contents); const ss = SpreadsheetApp.openById(SHEET_ID); const sh = ss.getSheetByName(d.sheet); if (!sh) return out({ok:false, err:'Sheet not found: '+d.sheet}); if (d.action === 'append') { sh.appendRow(d.row); return out({ok:true}); } if (d.action === 'delete') { sh.deleteRow(d.sheetRow); // 1-indexed sheet row number return out({ok:true}); } if (d.action === 'setCell') { sh.getRange(d.row, d.col).setValue(d.value); return out({ok:true}); } return out({ok:false, err:'Unknown action'}); } function out(obj) { return ContentService .createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); }
  1. Open your Sheet → Extensions → Apps Script
  2. Delete existing code, paste above, click Save (💾)
  3. Deploy → New deployment → Web app
  4. Execute as: Me · Who has access: Anyone
  5. Copy the Web App URL → paste into Settings → Apps Script URL
