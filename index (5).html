Copy this code into Google Apps Script to enable adding & deleting rows from the dashboard:

const SHEET_ID = '1RCZTOvlqDL14eaZlfS75zmkDnHo56NEYe5q3fm_U_gg'; function doPost(e) { const data = JSON.parse(e.postData.contents); const ss = SpreadsheetApp.openById(SHEET_ID); const sh = ss.getSheetByName(data.sheet); if (!sh) return out({ok:false, err:'Sheet not found'}); if (data.action === 'append') { sh.appendRow(data.row); return out({ok:true}); } if (data.action === 'delete') { sh.deleteRow(data.rowIndex + 2); // +2 skips header return out({ok:true}); } return out({ok:false, err:'Unknown action'}); } function out(obj) { return ContentService .createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); }
  1. Open your Google Sheet → Extensions → Apps Script
  2. Delete the default code, paste the code above, click Save
  3. Click Deploy → New deployment → Web app
  4. Execute as: Me · Who has access: Anyone
  5. Copy the Web App URL and paste it into Settings
